from fastapi import FastAPI, Request, Response, Depends, HTTPException from fastapi.middleware.cors import CORSMiddleware from fastapi.responses import HTMLResponse, JSONResponse from fastapi.templating import Jinja2Templates import sqlite3 import jwt import os import time import requests import uvicorn app = FastAPI() app.add_middleware( CORSMiddleware, allow_origin_regex=".*", allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) templates = Jinja2Templates(directory=".") # stockfishbot.html이 있는 폴더 SECRET_KEY = "원하시는_복잡한_비밀번호_입력" STOCKFISHBOT_USER_DB = "./database/stockfishbot_users/stockfishbot_users.db" KAKAO_QUERY_API = "http://127.0.0.1:3000/query" # JWT 검증 함수 def get_current_admin(request: Request): token = request.cookies.get("admin_token") if not token: raise HTTPException(status_code=401, detail="로그인이 필요합니다.") try: payload = jwt.decode(token, SECRET_KEY, algorithms=["HS256"]) return payload["user_id"] except: raise HTTPException(status_code=401, detail="유효하지 않은 세션입니다.") # 1. 메인 페이지 렌더링 @app.get("/", response_class=HTMLResponse) async def serve_dashboard(request: Request): return templates.TemplateResponse("stockfishbot.html", {"request": request}) # 2. 로그인 API (OTP 검증) @app.post("/api/login") async def login(request: Request, response: Response): data = await request.json() otp_input = data.get("otp") otp_path = f"./database/web_otp/otp_{otp_input}.txt" if not os.path.exists(otp_path): return JSONResponse(status_code=401, content={"error": "잘못된 OTP입니다."}) # 파일이 생성된(수정된) 시간 가져오기 file_time = os.path.getmtime(otp_path) with open(otp_path, "r") as f: # 파일 안의 내용을 읽어 user_id와 room_id로 분리 file_content = f.read().strip() user_id, room_id = file_content.split(',') os.remove(otp_path) # 사용한 OTP 폐기 # 3분(180초) 지났으면 만료 (파일 생성 시간 기준) if time.time() - file_time > 180: return JSONResponse(status_code=401, content={"error": "만료된 OTP입니다."}) # 로그인 성공! HttpOnly 쿠키에 토큰 저장 (보안) # user_id에 띄어쓰기가 들어갈 수 있으므로 strip()으로 공백 제거 token = jwt.encode({"user_id": user_id.strip(), "exp": time.time() + 7200}, SECRET_KEY, algorithm="HS256") res = JSONResponse(content={"message": "로그인 성공"}) res.set_cookie(key="admin_token", value=token, httponly=True, secure=True) return res # 3. [봇 DB] 전역 밴/관리자 리스트 불러오기 API @app.get("/api/admin/bot_users") async def get_bot_users(admin_id: str = Depends(get_current_admin)): with sqlite3.connect(STOCKFISHBOT_USER_DB) as conn: cur = conn.cursor() cur.execute("SELECT user_id, is_admin, is_banned, warnings FROM users") rows = cur.fetchall() result = [{"user_id": r[0], "is_admin": r[1], "is_banned": r[2], "warnings": r[3]} for r in rows] return {"users": result} # 4. [카톡 DB] 채팅 로그 불러오기 API (Port 3000 로컬 API 활용) @app.get("/api/admin/chatlogs/{room_id}/{user_id}") async def get_chatlogs(room_id: str, user_id: str, admin_id: str = Depends(get_current_admin)): payload = { "query": "SELECT message, created_at FROM chat_logs WHERE user_id = ? AND chat_id = ? ORDER BY id DESC LIMIT 50", "bind": [user_id, room_id] } try: res = requests.post(KAKAO_QUERY_API, json=payload, timeout=3) if res.status_code == 200: return res.json() except Exception as e: return JSONResponse(status_code=500, content={"error": str(e)}) if __name__ == "__main__": uvicorn.run(app, host="127.0.0.1", port=8000)
